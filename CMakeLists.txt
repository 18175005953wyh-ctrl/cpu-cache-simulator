cmake_minimum_required(VERSION 3.20)
project(CPUCacheSimulator VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(STRICT_WARNINGS "Treat warnings as errors" OFF)
add_library(cache_core src/cache.c src/trace.c src/report.c)
target_include_directories(cache_core PUBLIC include)
add_executable(cpu_cache_simulator src/main.c)
target_link_libraries(cpu_cache_simulator PRIVATE cache_core)
include(CTest)
if(BUILD_TESTING)
    add_executable(cache_tests tests/test_runner.c)
    target_link_libraries(cache_tests PRIVATE cache_core)
    add_test(NAME core_tests COMMAND cache_tests)
    add_test(NAME cli_tests COMMAND ${CMAKE_COMMAND}
        "-DAPP=$<TARGET_FILE:cpu_cache_simulator>"
        "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-fixtures"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.cmake")
endif()
foreach(target cache_core cpu_cache_simulator cache_tests)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /utf-8)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
            if(STRICT_WARNINGS)
                target_compile_options(${target} PRIVATE /WX)
            endif()
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
            if(STRICT_WARNINGS)
                target_compile_options(${target} PRIVATE -Werror)
            endif()
        endif()
    endif()
endforeach()
set_target_properties(cpu_cache_simulator PROPERTIES
    VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    VS_DEBUGGER_COMMAND_ARGUMENTS "--cache-size 64 --block-size 16 --associativity 1 --trace data/sample_trace.txt")
